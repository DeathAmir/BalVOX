cmake_minimum_required(VERSION 3.20)
project(BalVOX VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5)
FetchContent_MakeAvailable(raylib)
add_executable(BalVOX src/main.cpp)
target_link_libraries(BalVOX PRIVATE raylib)
target_compile_definitions(BalVOX PRIVATE ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(WIN32)
  target_link_libraries(BalVOX PRIVATE winmm)
endif()
if(MSVC)
  target_compile_options(BalVOX PRIVATE /W4 /permissive-)
else()
  target_compile_options(BalVOX PRIVATE -Wall -Wextra -Wpedantic)
endif()
